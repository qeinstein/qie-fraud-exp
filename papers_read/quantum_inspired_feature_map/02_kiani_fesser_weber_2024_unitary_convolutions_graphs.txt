Standard graph convolution layers evaluate:

or D^{-1} A). Because P has eigenvalues bounded in [-1, 1] with an isolated
eigenvalue at 1 corresponding to the trivial constant subspace, repeated
application of P acts as a strict contraction mapping:
    lim_{l -> infty} P^l X = 1 * v^T (constant vector)

import numpy as np

def unitary_spectral_feature_map(H: np.ndarray, raw_features: np.ndarray, 
                                 times: list = [0.5, 1.0, 2.0]) -> np.ndarray:
    """
    Applies Kiani et al. unitary convolution feature map:
    Z(t) = exp(-i * H * t) @ (raw_features)
    Preserves L2 norms and extracts both amplitude and phase features.
    """
    w, V = np.linalg.eigh(H)
    feature_blocks = [raw_features]
    
    # Cast raw features to complex
    Z_0 = raw_features.astype(np.complex128)
    
    for t in times:
        # Diagonal unitary operator in eigenbasis
        exp_diag = np.exp(-1j * w * t)
        # U(t) @ Z_0 = V @ diag(exp_diag) @ V^\dagger @ Z_0
        Z_t = V @ (exp_diag[:, None] * (V.conj().T @ Z_0))
        
        # Extract norm-preserving magnitude and interference phase
        magnitude = np.abs(Z_t)
        phase = np.angle(Z_t)
        
        feature_blocks.append(magnitude)
        feature_blocks.append(phase)
        
    return np.hstack(feature_blocks)
