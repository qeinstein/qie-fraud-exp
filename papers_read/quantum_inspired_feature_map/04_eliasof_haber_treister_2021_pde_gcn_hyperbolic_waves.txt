       partial_t u = div( K grad u )

2. Hyperbolic PDEs (Wave Equations - Wave GCNs):

import numpy as np

def compute_wave_energy_densities(H: np.ndarray, psi_t: np.ndarray) -> tuple:
    """
    Computes hyperbolic wave kinetic and potential energy densities per node:
    E_kin[i] = |(H @ psi_t)[i]|^2
    E_pot[i] = sum_j |H_{ij}| * |psi_i - psi_j|^2
    """
    N = H.shape[0]
    # Time derivative: d psi / dt = -i * H @ psi
    d_psi_dt = -1j * (H @ psi_t)
    E_kin = np.abs(d_psi_dt) ** 2
    
    # Potential energy via local gradient
    E_pot = np.zeros(N)
    for i in range(N):
        diffs = psi_t[i] - psi_t
        weights = np.abs(H[i, :])
        E_pot[i] = np.sum(weights * np.abs(diffs) ** 2)
        
    return E_kin, E_pot
